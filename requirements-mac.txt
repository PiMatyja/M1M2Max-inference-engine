# Python packages for the server and the pack tools, built from source archives only (no wheels).
# Every line is pinned to one version and its sdist SHA-256 from PyPI, so pip refuses anything else.
# scripts/python-env.sh installs them into a venv made from Homebrew python 3.14: it downloads every archive with
#   .venv/bin/pip download --no-binary :all: --no-deps --require-hashes -r requirements-mac.txt -d .venv/sdists
# then installs them one at a time, in the order below, with --no-deps --no-build-isolation --no-index.
# --no-build-isolation means the build backends (setuptools, flit-core) must already be in the venv, so they come first.
# pip itself must be new enough to know Python 3.14 (pip 25.2 or newer; Homebrew python 3.14 ships one).

# Build backend for markupsafe, pyyaml and regex.
